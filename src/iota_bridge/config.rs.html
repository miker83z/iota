40#[serde(rename_all = "kebab-case")]
41pub struct EthConfig {
42    /// Rpc url for Eth fullnode, used for query stuff.
43    pub eth_rpc_url: String,
44    /// The proxy address of IotaBridge
45    pub eth_bridge_proxy_address: String,
46    /// The expected BridgeChainId on Eth side.
47    pub eth_bridge_chain_id: u8,
48    /// The starting block for EthSyncer to monitor eth contracts.
49    /// It is required when `run_client` is true. Usually this is
50    /// the block number when the bridge contracts are deployed.
51    /// When BridgeNode starts, it reads the contract watermark from storage.
52    /// If the watermark is not found, it will start from this fallback block
53    /// number. If the watermark is found, it will start from the watermark.
54    /// this v.s.`eth_contracts_start_block_override`:
55    pub eth_contracts_start_block_fallback: Option<u64>,
56    /// The starting block for EthSyncer to monitor eth contracts. It overrides
57    /// the watermark in storage. This is useful when we want to reprocess the
58    /// events from a specific block number.
59    /// Note: this field has to be reset after starting the BridgeNode,
60    /// otherwise it will reprocess the events from this block number every
61    /// time it starts.
62    #[serde(skip_serializing_if = "Option::is_none")]
63    pub eth_contracts_start_block_override: Option<u64>,
64}

69pub struct IotaConfig {
70    /// Rpc url for IOTA fullnode, used for query stuff and submit transactions.
71    pub iota_rpc_url: String,
72    /// The expected BridgeChainId on IOTA side.
73    pub iota_bridge_chain_id: u8,
74    /// Path of the file where bridge client key (any IotaKeyPair) is stored.
75    /// If `run_client` is true, and this is None, then use
76    /// `bridge_authority_key_path` as client key.
77    #[serde(skip_serializing_if = "Option::is_none")]
78    pub bridge_client_key_path: Option<PathBuf>,
79    /// The gas object to use for paying for gas fees for the client. It needs
80    /// to be owned by the address associated with bridge client key. If not
81    /// set and `run_client` is true, it will query and use the gas object
82    /// with highest amount for the account.
83    #[serde(skip_serializing_if = "Option::is_none")]
84    pub bridge_client_gas_object: Option<ObjectID>,
85    /// Override the last processed EventID for bridge module `bridge`.
86    /// When set, IotaSyncer will start from this cursor (exclusively) instead
87    /// of the one in storage. If the cursor is not found in storage or
88    /// override, the query will start from genesis. Key: iota module,
89    /// Value: last processed EventID (tx_digest, event_seq). Note 1: This
90    /// field should be rarely used. Only use it when you understand how to
91    /// follow up. Note 2: the EventID needs to be valid, namely it must
92    /// exist and matches the filter. Otherwise, it will miss one event
93    /// because of fullnode Event query semantics.
94    #[serde(skip_serializing_if = "Option::is_none")]
95    pub iota_bridge_module_last_processed_event_id_override: Option<EventID>,
96}

112    #[serde(skip_serializing_if = "Option::is_none")]

117    /// IOTA configuration

119    /// Eth configuration
120    pub eth: EthConfig,

125impl BridgeNodeConfig {
126    pub async fn validate(
127        &self,
128        metrics: Arc<BridgeMetrics>,
129    ) -> anyhow::Result<(BridgeServerConfig, Option<BridgeClientConfig>)> {
130        if !is_route_valid(
131            BridgeChainId::try_from(self.iota.iota_bridge_chain_id)?,
132            BridgeChainId::try_from(self.eth.eth_bridge_chain_id)?,
133        ) {
134            bail!(
135                "Route between IOTA chain id {} and Eth chain id {} is not valid",
136                self.iota.iota_bridge_chain_id,
137                self.eth.eth_bridge_chain_id,
138            );
139        };
140
141        let bridge_authority_key = match read_key(&self.bridge_authority_key_path, true)? {
142            IotaKeyPair::Secp256k1(key) => key,
143            _ => unreachable!("we required secp256k1 key in `read_key`"),
144        };
145
146        // we do this check here instead of `prepare_for_iota` below because
147        // that is only called when `run_client` is true.
148        let iota_client =
149            Arc::new(IotaClient::<IotaSdkClient>::new(&self.iota.iota_rpc_url).await?);
150        let bridge_committee = iota_client
151            .get_bridge_committee()
152            .await
153            .map_err(|e| anyhow!("Error getting bridge committee: {:?}", e))?;
154        if !bridge_committee.is_active_member(&bridge_authority_key.public().into()) {
155            bail!("Bridge authority key is not part of bridge committee");
156        }
157
158        let (eth_client, eth_contracts) = self.prepare_for_eth(metrics).await?;
159        let bridge_summary = iota_client
160            .get_bridge_summary()
161            .await
162            .map_err(|e| anyhow!("Error getting bridge summary: {:?}", e))?;
163        if bridge_summary.chain_id != self.iota.iota_bridge_chain_id {
164            bail!(
165                "Bridge chain id mismatch: expected {}, but connected to {}",
166                self.iota.iota_bridge_chain_id,
167                bridge_summary.chain_id
168            );
169        }
170
171        // Validate approved actions that must be governace actions
172        for action in &self.approved_governance_actions {
173            if !action.is_governace_action() {
174                bail!(
175                    "{:?}",
176                    BridgeError::ActionIsNotGovernanceAction(action.clone())
177                );
178            }
179        }
180        let approved_governance_actions = self.approved_governance_actions.clone();
181
182        let bridge_server_config = BridgeServerConfig {
183            key: bridge_authority_key,
184            metrics_port: self.metrics_port,
185            server_listen_port: self.server_listen_port,
186            iota_client: iota_client.clone(),
187            eth_client: eth_client.clone(),
188            approved_governance_actions,
189        };
190        if !self.run_client {
191            return Ok((bridge_server_config, None));
192        }
193
194        // If client is enabled, prepare client config
195        let (bridge_client_key, client_iota_address, gas_object_ref) =
196            self.prepare_for_iota(iota_client.clone()).await?;
197
198        let db_path = self
199            .db_path
200            .clone()
201            .ok_or(anyhow!("`db_path` is required when `run_client` is true"))?;
202
203        let bridge_client_config = BridgeClientConfig {
204            iota_address: client_iota_address,
205            key: bridge_client_key,
206            gas_object_ref,
207            metrics_port: self.metrics_port,
208            iota_client: iota_client.clone(),
209            eth_client: eth_client.clone(),
210            db_path,
211            eth_contracts,
212            // in `prepare_for_eth` we check if this is None when `run_client` is true. Safe to
213            // unwrap here.
214            eth_contracts_start_block_fallback: self
215                .eth
216                .eth_contracts_start_block_fallback
217                .unwrap(),
218            eth_contracts_start_block_override: self.eth.eth_contracts_start_block_override,
219            iota_bridge_module_last_processed_event_id_override: self
220                .iota
221                .iota_bridge_module_last_processed_event_id_override,
222        };
223
224        Ok((bridge_server_config, Some(bridge_client_config)))
225    }

227    async fn prepare_for_eth(
228        &self,
229        metrics: Arc<BridgeMetrics>,
230    ) -> anyhow::Result<(Arc<EthClient<MeteredEthHttpProvider>>, Vec<EthAddress>)> {
231        let bridge_proxy_address = EthAddress::from_str(&self.eth.eth_bridge_proxy_address)?;
232        let provider = Arc::new(
233            new_metered_eth_provider(&self.eth.eth_rpc_url, metrics.clone())
234                .unwrap()
235                .interval(std::time::Duration::from_millis(2000)),
236        );
237        let chain_id = provider.get_chainid().await?;
238        let (committee_address, limiter_address, vault_address, config_address) =
239            get_eth_contract_addresses(bridge_proxy_address, &provider).await?;
240        let config = EthBridgeConfig::new(config_address, provider.clone());
241
242        if self.run_client && self.eth.eth_contracts_start_block_fallback.is_none() {
243            bail!("eth_contracts_start_block_fallback is required when run_client is true");
244        }
245
246        // If bridge chain id is Eth Mainent or Sepolia, we expect to see chain
247        // identifier to match accordingly.
248        let bridge_chain_id: u8 = config.chain_id().call().await?;
249        if self.eth.eth_bridge_chain_id != bridge_chain_id {
250            bail!(
251                "Bridge chain id mismatch: expected {}, but connected to {}",
252                self.eth.eth_bridge_chain_id,
253                bridge_chain_id
254            );
255        }
256        if bridge_chain_id == BridgeChainId::EthMainnet as u8 && chain_id.as_u64() != 1 {
257            bail!(
258                "Expected Eth chain id 1, but connected to {}",
259                chain_id.as_u64()
260            );
261        }
262        if bridge_chain_id == BridgeChainId::EthSepolia as u8 && chain_id.as_u64() != 11155111 {
263            bail!(
264                "Expected Eth chain id 11155111, but connected to {}",
265                chain_id.as_u64()
266            );
267        }
268        info!(
269            "Connected to Eth chain: {}, Bridge chain id: {}",
270            chain_id.as_u64(),
271            bridge_chain_id,
272        );
273
274        let eth_client = Arc::new(
275            EthClient::<MeteredEthHttpProvider>::new(
276                &self.eth.eth_rpc_url,
277                HashSet::from_iter(vec![
278                    bridge_proxy_address,
279                    committee_address,
280                    config_address,
281                    limiter_address,
282                    vault_address,
283                ]),
284                metrics,
285            )
286            .await?,
287        );
288        let contract_addresses = vec![
289            bridge_proxy_address,
290            committee_address,
291            config_address,
292            limiter_address,
293            vault_address,
294        ];
295        Ok((eth_client, contract_addresses))
296    }

298    async fn prepare_for_iota(
299        &self,
300        iota_client: Arc<IotaClient<IotaSdkClient>>,
301    ) -> anyhow::Result<(IotaKeyPair, IotaAddress, ObjectRef)> {
302        let bridge_client_key = match &self.iota.bridge_client_key_path {
303            None => read_key(&self.bridge_authority_key_path, true),
304            Some(path) => read_key(path, false),
305        }?;
306
307        // If bridge chain id is IOTA Mainent or Testnet, we expect to see chain
308        // identifier to match accordingly.
309        let iota_identifier = iota_client
310            .get_chain_identifier()
311            .await
312            .map_err(|e| anyhow!("Error getting chain identifier from IOTA: {:?}", e))?;
313        if self.iota.iota_bridge_chain_id == BridgeChainId::IotaMainnet as u8
314            && iota_identifier != get_mainnet_chain_identifier().to_string()
315        {
316            bail!(
317                "Expected iota chain identifier {}, but connected to {}",
318                self.iota.iota_bridge_chain_id,
319                iota_identifier
320            );
321        }
322        if self.iota.iota_bridge_chain_id == BridgeChainId::IotaTestnet as u8
323            && iota_identifier != get_testnet_chain_identifier().to_string()
324        {
325            bail!(
326                "Expected iota chain identifier {}, but connected to {}",
327                self.iota.iota_bridge_chain_id,
328                iota_identifier
329            );
330        }
331        info!(
332            "Connected to IOTA chain: {}, Bridge chain id: {}",
333            iota_identifier, self.iota.iota_bridge_chain_id,
334        );
335
336        let client_iota_address = IotaAddress::from(&bridge_client_key.public());
337
338        // TODO: decide a minimal amount here
339        let gas_object_id = match self.iota.bridge_client_gas_object {
340            Some(id) => id,
341            None => {
342                let iota_client = IotaClientBuilder::default()
343                    .build(&self.iota.iota_rpc_url)
344                    .await?;
345                let coin =
346                    pick_highest_balance_coin(iota_client.coin_read_api(), client_iota_address, 0)
347                        .await?;
348                coin.coin_object_id
349            }
350        };
351        let (gas_coin, gas_object_ref, owner) = iota_client
352            .get_gas_data_panic_if_not_gas(gas_object_id)
353            .await;
354        if owner != Owner::AddressOwner(client_iota_address) {
355            bail!(
356                "Gas object {:?} is not owned by bridge client key's associated iota address {:?}, but {:?}",
357                gas_object_id,
358                client_iota_address,
359                owner
360            );
361        }
362        info!(
363            "Starting bridge client with address: {:?}, gas object {:?}, balance: {}",
364            client_iota_address,
365            gas_object_ref.0,
366            gas_coin.value()
367        );
368
369        Ok((bridge_client_key, client_iota_address, gas_object_ref))
370    }

409pub async fn pick_highest_balance_coin(
410    coin_read_api: &CoinReadApi,
411    address: IotaAddress,
412    minimal_amount: u64,
413) -> anyhow::Result<Coin> {
414    let mut highest_balance = 0;
415    let mut highest_balance_coin = None;
416    coin_read_api
417        .get_coins_stream(address, None)
418        .for_each(|coin: Coin| {
419            if coin.balance > highest_balance {
420                highest_balance = coin.balance;
421                highest_balance_coin = Some(coin.clone());
422            }
423            future::ready(())
424        })
425        .await;
426    if highest_balance_coin.is_none() {
427        bail!("No IOTA coins found for address {:?}", address);
428    }
429    if highest_balance < minimal_amount {
430        bail!(
431            "Found no single coin that has >= {} balance IOTA for address {:?}",
432            minimal_amount,
433            address,
434        );
435    }
436    Ok(highest_balance_coin.unwrap())
437}
